  "startups": [
    {
      "name": "string",
      "summary": "One sentence summary",
      "url": "the post's URL, copied exactly from the list",
      "source": "Hacker News",
      "problem": "What problem it solves",
      "businessModel": "How it makes money",
      "keyBet": "The core assumption that must be true for this to work"
    }
  ]
